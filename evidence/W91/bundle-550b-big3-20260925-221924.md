630: /// Outcome of waiting for `jarvis serve` to become healthy.
631: ///
632: /// Unlike [`wait_for_url`] this differentiates "server is up but degraded"
633: /// (HTTP 503 — usually inference engine failed to load) from "server never
634: /// came up" and from "child process died before serving anything", because
635: /// each needs a different user-facing message.
636: #[derive(Debug)]
637: enum JarvisStartResult {
638:     /// `/health` returned 2xx.
639:     Ready,
640:     /// Server replied 503. The body is the actionable message (typically
641:     /// "engine not ready" or a model-load error).
642:     ServiceUnavailable(String),
643:     /// The `jarvis serve` child exited before `/health` returned 2xx.
644:     EarlyExit { code: Option<i32>, stderr: String },
645:     /// Deadline elapsed without ever seeing 2xx or an early exit.
646:     Timeout,
647: }

649: /// Spawn a detached task that continuously drains `jarvis serve`'s
650: /// stderr into a rolling tail buffer.
651: ///
652: /// We MUST keep reading stderr for as long as the child runs — `jarvis
653: /// serve` is chatty (engine load progress, request logs), and the OS
654: /// pipe buffer is small (4 KB on Windows, 64 KB on Linux). Once full,
655: /// the child's next stderr write blocks indefinitely and the server
656: /// hangs mid-operation. The drainer reads in chunks and keeps only the
657: /// last `STDERR_TAIL_LIMIT` bytes — enough to surface a tail trace if
658: /// the child later dies, without unbounded memory growth.
659: ///
660: /// Returns immediately after spawning the task; the task ends naturally
661: /// when the child closes stderr (i.e. exits).
662: fn spawn_jarvis_stderr_drainer(mut stderr: tokio::process::ChildStderr, tail: StderrTail) {
663:     use tokio::io::AsyncReadExt;
664:     tokio::spawn(async move {
665:         let mut buf = vec![0u8; 4096];
666:         loop {
667:             match stderr.read(&mut buf).await {
668:                 Ok(0) => break,  // EOF — child closed stderr
669:                 Err(_) => break, // pipe broke — also done
670:                 Ok(n) => {
671:                     let mut t = tail.lock().await;
672:                     t.extend_from_slice(&buf[..n]);
673:                     if t.len() > STDERR_TAIL_LIMIT {
674:                         let drop_n = t.len() - STDERR_TAIL_LIMIT;
675:                         t.drain(..drop_n);
676:                     }
677:                 }
678:             }
679:         }
680:     });
681: }

683: /// Read whatever the stderr drainer has buffered so far.

685: /// Safe to call at any time; returns an empty string before the
686: /// drainer has seen any bytes. Trimmed.

688:     let tail = backend.lock().await.jarvis_stderr_tail.clone();

693: /// Poll `jarvis serve` health, watching the child process state so we
694: /// never wait 10 minutes for a process that crashed in the first second.
695: async fn wait_for_jarvis_health(
696:     url: &str,
697:     timeout: Duration,
698:     backend: &SharedBackend,
699: ) -> JarvisStartResult {
700:     let client = match reqwest::Client::builder()
701:         .timeout(Duration::from_secs(2))
702:         .build()
703:     {
704:         Ok(c) => c,
705:         Err(_) => return JarvisStartResult::Timeout,
706:     };
707:     let deadline = tokio::time::Instant::now() + timeout;
708:     loop {
709:         // 1. Has the child already exited? `try_wait` is non-blocking; on
710:         // Windows where uv / python / the Rust extension can fail to load
711:         // very fast, this catches the crash within ~500ms instead of after
712:         // the full HTTP timeout window.
713:         let exit_status = {
714:             let mut mgr = backend.lock().await;
715:             match mgr.jarvis.as_mut() {
716:                 Some(h) => h.child.try_wait().ok().flatten(),
717:                 None => None,
718:             }
719:         };
720:         if let Some(status) = exit_status {
721:             let stderr = read_jarvis_stderr_tail(backend).await;
722:             return JarvisStartResult::EarlyExit {
723:                 code: status.code(),
724:                 stderr,
725:             };
726:         }
727: 
728:         // 2. Try the health endpoint.
729:         match client.get(url).send().await {
730:             Ok(resp) => {
731:                 let status = resp.status();
732:                 if status.is_success() {
733:                     return JarvisStartResult::Ready;
734:                 }
735:                 if status == reqwest::StatusCode::SERVICE_UNAVAILABLE {
736:                     // Server is up but the inference engine is not. This
737:                     // is a terminal-for-us state — polling won't change
738:                     // anything; the user has to fix their engine config.
739:                     let body = resp.text().await.unwrap_or_default();
740:                     return JarvisStartResult::ServiceUnavailable(body);
741:                 }
742:                 // Other non-2xx (e.g. 404 during a brief routing-table
743:                 // warmup window) — fall through and keep polling.
744:             }
745:             Err(_) => {
746:                 // Connection refused / DNS / timeout — server still
747:                 // booting. Keep polling.
748:             }
749:         }
750: 
751:         if tokio::time::Instant::now() >= deadline {
752:             return JarvisStartResult::Timeout;
753:         }
754:         tokio::time::sleep(Duration::from_millis(500)).await;
755:     }
756: }

2277: async fn run_jarvis_command(args: Vec<String>) -> Result<String, String> {
2278:     let uv_bin = resolve_bin("uv");
2279: 
2280:     let mut cmd_args = vec!["run".to_string(), "jarvis".to_string()];
2281:     cmd_args.extend(args.iter().cloned());
2282: 
2283:     let mut cmd = tokio::process::Command::new(&uv_bin);
2284:     cmd.args(&cmd_args);
2285:     // Run from the project root so `uv run jarvis` resolves the OpenJarvis
2286:     // project regardless of the app's launch cwd. In a packaged install the
2287:     // cwd isn't the checkout, so without this `jarvis` isn't found and the
2288:     // backend never starts — the UI then shows "Failed to get response"
2289:     // (see #531).
2290:     if let Some(ref root) = find_project_root() {
2291:         cmd.current_dir(root);
2292:     }
2293: 
2294:     let is_serve = args.first().map(|a| a.as_str() == "serve").unwrap_or(false);
2295: 
2296:     if !is_serve {
2297:         // Short-lived command (e.g. `stop`, `status`): wait for it and return
2298:         // its captured output.
2299:         let output = cmd
2300:             .output()
2301:             .await
2302:             .map_err(|e| format!("Failed to launch jarvis: {}", e))?;
2303:         return if output.status.success() {
2304:             Ok(String::from_utf8_lossy(&output.stdout).to_string())
2305:         } else {
2306:             Err(String::from_utf8_lossy(&output.stderr).to_string())
2307:         };
2308:     }
2309: 
2310:     // `jarvis serve` is a long-running server that never exits. The old code
2311:     // used `.output()`, which waits for the process to exit and so hung this
2312:     // command forever — the "Start" button never resolved (#531). Spawn it
2313:     // detached instead, drain stderr (a full 4 KB Windows pipe can otherwise
2314:     // stall the child mid-startup, #309), and poll /health for readiness.
2315:     cmd.stdout(std::process::Stdio::null())
2316:         .stderr(std::process::Stdio::piped());
2317:     let mut child = cmd
2318:         .spawn()
2319:         .map_err(|e| format!("Failed to launch jarvis serve: {}", e))?;
2320: 
2321:     let tail: StderrTail = Arc::new(Mutex::new(Vec::new()));
2322:     if let Some(stderr) = child.stderr.take() {
2323:         spawn_jarvis_stderr_drainer(stderr, tail.clone());
2324:     }
2325: 
2326:     let client = reqwest::Client::builder()
2327:         .timeout(Duration::from_secs(2))
2328:         .build()
2329:         .map_err(|e| format!("Failed to build HTTP client: {}", e))?;
2330:     let url = format!("http://127.0.0.1:{}/health", JARVIS_PORT);
2331:     let deadline = tokio::time::Instant::now() + Duration::from_secs(120);
2332: 
2333:     loop {
2334:         // Surface an early crash (bad venv, missing Rust ext, etc.) right away
2335:         // instead of waiting out the full readiness timeout.
2336:         if let Ok(Some(status)) = child.try_wait() {
2337:             let stderr = String::from_utf8_lossy(tail.lock().await.as_slice()).into_owned();
2338:             return Err(format!(
2339:                 "jarvis serve exited (code {:?}) before becoming healthy:\n{}",
2340:                 status.code(),
2341:                 stderr.trim()
2342:             ));
2343:         }
2344:         if let Ok(resp) = client.get(&url).send().await {
2345:             if resp.status().is_success() {
2346:                 // Leave the server running (the Child is detached on drop —
2347:                 // kill_on_drop defaults to false); `stop` tears it down.
2348:                 return Ok(format!(
2349:                     "jarvis serve is ready on http://127.0.0.1:{}",
2350:                     JARVIS_PORT
2351:                 ));
2352:             }
2353:         }
2354:         if tokio::time::Instant::now() >= deadline {
2355:             return Err(format!(
2356:                 "jarvis serve did not become healthy on port {} within 120s.",
2357:                 JARVIS_PORT
2358:             ));
2359:         }
2360:         tokio::time::sleep(Duration::from_millis(500)).await;
2361:     }
2362: }
